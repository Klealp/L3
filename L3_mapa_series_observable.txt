// =====================================================================
// LAB 3 – Mapa de Colombia + 4 series de tiempo (cobertura neta)
// Pegar cada bloque en una CELDA distinta de Observable.
// Archivos a adjuntar en Observable (Files / clip 📎):
//   1. cobertura_neta.csv        (generado desde TerriData_Dim4.xlsx)
//   2. MGN_ANM_MPIOS.geojson     (municipios)
//   3. MGN_ANM_DPTOS.geojson     (departamentos, solo para los bordes)
// =====================================================================


// ---------------------------------------------------------------------
// CELDA 1 – Datos de cobertura
// Se lee como texto (sin typed) para NO perder el cero de "05001".
// ---------------------------------------------------------------------
datos = {
  const num = v => (v === "" ? null : +v);   // "" -> null, "87.3" -> 87.3
  const filas = await FileAttachment("cobertura_neta.csv").csv();
  return filas.map(d => ({
    nivel: d.nivel,                // "municipio" | "departamento"
    cod_dpto: d.cod_dpto,          // "05"
    departamento: d.departamento,
    cod_entidad: d.cod_entidad,    // "05001" (igual a MPIO_CDPMP del GeoJSON)
    entidad: d.entidad,
    anio: +d.anio,
    total: num(d.total),
    transicion: num(d.transicion),
    primaria: num(d.primaria),
    secundaria: num(d.secundaria),
    media: num(d.media)
  }));
}


// ---------------------------------------------------------------------
// CELDA 2 – GeoJSON de municipios
// ---------------------------------------------------------------------
mpios = FileAttachment("MGN_ANM_MPIOS.geojson").json()


// ---------------------------------------------------------------------
// CELDA 3 – GeoJSON de departamentos
// ---------------------------------------------------------------------
dptos = FileAttachment("MGN_ANM_DPTOS.geojson").json()


// ---------------------------------------------------------------------
// CELDA 4 – Tablero (mapa 40% + grilla 4x1 de series)
// ---------------------------------------------------------------------
tablero = {
  // ===== 1. Parámetros =====
  const AZUL = "#003893", ROJO = "#CE1126", GRIS = "#c4c4c4";
  const W = 1400;                                   // ancho de la celda
  const mapaW = W * 0.40;                            // 40% para el mapa
  // Colombia continental (sin San Andrés, código 88): define el encuadre del mapa
  const continente = {type: "FeatureCollection",
    features: mpios.features.filter(f => f.properties.DPTO_CCDGO !== "88")};
  const islas = {type: "FeatureCollection",
    features: mpios.features.filter(f => f.properties.DPTO_CCDGO === "88")};
  // Alto que NECESITA el mapa para ocupar ese ancho (según su proporción)
  const [[bx0, by0], [bx1, by1]] = d3.geoPath(d3.geoMercator().fitWidth(100, continente)).bounds(continente);
  const mapaH = (mapaW - 10) * (by1 - by0) / (bx1 - bx0);
  // Alto total = mapa + márgenes (45 arriba, 10 abajo), sin pasar el alto de la pantalla
  const H = Math.min(mapaH + 55, window.innerHeight - 90);
  const gx0 = mapaW + 40, gx1 = W - 15;              // zona de los gráficos
  const niveles = [
    {key: "transicion", titulo: "Transición"},
    {key: "primaria",   titulo: "Primaria"},
    {key: "secundaria", titulo: "Secundaria"},
    {key: "media",      titulo: "Media"}
  ];
  const inicial = "11001";                           // Bogotá al cargar

  // ===== 2. Preparar datos (equivale a un groupby de pandas) =====
  const munis  = datos.filter(d => d.nivel === "municipio");
  const deptos = datos.filter(d => d.nivel === "departamento");
  const porAnio = (a, b) => a.anio - b.anio;

  const seriesMuni  = d3.group(munis,  d => d.cod_entidad);  // Map código -> filas
  const seriesDepto = d3.group(deptos, d => d.cod_dpto);
  for (const s of seriesMuni.values())  s.sort(porAnio);
  for (const s of seriesDepto.values()) s.sort(porAnio);

  // Serie del departamento: fila oficial "XX000" de TerriData.
  // Bogotá no tiene esa fila -> promedio simple de sus municipios.
  function serieDepto(cod) {
    if (seriesDepto.has(cod)) return seriesDepto.get(cod);
    return d3.rollups(munis.filter(d => d.cod_dpto === cod),
                      v => Object.fromEntries(niveles.map(n => [n.key, d3.mean(v, d => d[n.key])])),
                      d => d.anio)
      .map(([anio, vals]) => ({anio, ...vals}))
      .sort(porAnio);
  }

  // Valor "Total" del último año disponible -> color del mapa
  const ultimoAnio = d3.max(munis.filter(d => d.total != null), d => d.anio);
  const totalUlt = new Map(munis.filter(d => d.anio === ultimoAnio && d.total != null)
                                .map(d => [d.cod_entidad, d.total]));
  const infoMuni = new Map(munis.map(d => [d.cod_entidad, d]));  // nombre y depto

  const vals = [...totalUlt.values()];
  const color = d3.scaleSequential(d3.interpolateYlGn)
    .domain([d3.quantile(vals, 0.02), d3.quantile(vals, 0.98)])  // sin outliers
    .clamp(true);

  // ===== 3. Corregir orden de los anillos (si el GeoJSON viene "al revés",
  //          D3 pintaría el planeta entero en vez del polígono) =====
  const invertir = g => g.type === "Polygon"
    ? g.coordinates.forEach(r => r.reverse())
    : g.coordinates.forEach(p => p.forEach(r => r.reverse()));
  for (const f of [...mpios.features, ...dptos.features])
    if (d3.geoArea(f) > 2 * Math.PI) invertir(f.geometry);

  const featMuni  = new Map(mpios.features.map(f => [f.properties.MPIO_CDPMP, f]));
  const featDepto = new Map(dptos.features.map(f => [f.properties.DPTO_CCDGO, f]));

  // ===== 4. SVG base =====
  const svg = d3.create("svg")
    .attr("width", W).attr("height", H)
    .attr("viewBox", [0, 0, W, H])
    .style("background", "white")
    .style("font", "11px sans-serif");

  // ===== 5. MAPA =====
  const proj = d3.geoMercator().fitExtent([[5, 45], [mapaW - 5, H - 10]], continente);
  const path = d3.geoPath(proj);

  // Recuadro (inset) para San Andrés y Providencia, arriba a la izquierda
  const ins = {x: 5, y: 50, w: mapaW * 0.16, h: mapaW * 0.22};
  const pathIslas = d3.geoPath(d3.geoMercator()
    .fitExtent([[ins.x + 6, ins.y + 6], [ins.x + ins.w - 6, ins.y + ins.h - 6]], islas));
  // Cada feature usa la proyección que le corresponde
  const pathMapa = f => (f.properties.DPTO_CCDGO === "88" ? pathIslas : path)(f);

  svg.append("text").attr("x", 5).attr("y", 18)
    .style("font-weight", "bold").style("font-size", "13px")
    .text(`Cobertura neta en educación – Total (${ultimoAnio})`);
  const etiqueta = svg.append("text").attr("x", 5).attr("y", 36)
    .attr("fill", ROJO).style("font-size", "12px");

  // Dibuja en el grupo g: municipios coloreados, bordes de departamentos y resaltados.
  // vector-effect "non-scaling-stroke" mantiene el grosor de las líneas al hacer zoom.
  function dibujarMapa(g, mf, df) {
    g.append("g")
      .selectAll("path")
      .data(mf)
      .join("path")
        .attr("d", pathMapa)
        .attr("fill", f => {
          const v = totalUlt.get(f.properties.MPIO_CDPMP);
          return v == null ? "#eeeeee" : color(v);
        })
        .attr("stroke", "white").attr("stroke-width", 0.2)
        .style("vector-effect", "non-scaling-stroke")
        .style("cursor", "pointer")
        .on("click", (event, f) => actualizar(f.properties.MPIO_CDPMP))
      .append("title")
        .text(f => {
          const v = totalUlt.get(f.properties.MPIO_CDPMP);
          return `${f.properties.MPIO_CNMBR}\nTotal ${ultimoAnio}: ${v == null ? "sin dato" : v.toFixed(1) + "%"}`;
        });

    // Bordes de departamentos (no reciben clics)
    g.append("g")
      .selectAll("path")
      .data(df)
      .join("path")
        .attr("d", pathMapa)
        .attr("fill", "none").attr("stroke", "#888").attr("stroke-width", 0.4)
        .style("vector-effect", "non-scaling-stroke")
        .style("pointer-events", "none");

    // Resaltados de la selección (encima de todo)
    return {
      deptoSel: g.append("path").attr("fill", "none").attr("stroke", AZUL)
        .attr("stroke-width", 1.5).style("vector-effect", "non-scaling-stroke")
        .style("pointer-events", "none"),
      muniSel: g.append("path").attr("fill", "none").attr("stroke", ROJO)
        .attr("stroke-width", 2.5).style("vector-effect", "non-scaling-stroke")
        .style("pointer-events", "none")
    };
  }

  // Zona del mapa continental con zoom y desplazamiento (recortada a su área)
  const clipMapa = DOM.uid("clipMapa");
  svg.append("clipPath").attr("id", clipMapa.id)
    .append("rect").attr("x", 0).attr("y", 40).attr("width", mapaW).attr("height", H - 40);
  const zonaMapa = svg.append("g").attr("clip-path", clipMapa);
  zonaMapa.append("rect").attr("x", 0).attr("y", 40)       // fondo para poder arrastrar en el vacío
    .attr("width", mapaW).attr("height", H - 40).attr("fill", "white");
  const gZoom = zonaMapa.append("g");
  const selCont = dibujarMapa(gZoom, continente.features, dptos.features.filter(f => f.properties.DPTO_CCDGO !== "88"));

  const zoom = d3.zoom()
    .scaleExtent([1, 15])
    .extent([[0, 40], [mapaW, H]]).translateExtent([[0, 40], [mapaW, H]])
    .on("zoom", e => gZoom.attr("transform", e.transform));
  zonaMapa.call(zoom);

  // Recuadro de San Andrés y Providencia: queda fijo (no se afecta por el zoom)
  svg.append("rect").attr("x", ins.x).attr("y", ins.y)
    .attr("width", ins.w).attr("height", ins.h)
    .attr("fill", "white").attr("stroke", "#bbb");
  svg.append("text").attr("x", ins.x).attr("y", ins.y + ins.h + 12)
    .style("font-size", "9px").text("San Andrés y Providencia");
  const selIslas = dibujarMapa(svg.append("g"), islas.features, dptos.features.filter(f => f.properties.DPTO_CCDGO === "88"));

  // Botones de zoom: acercar, alejar y restablecer (también sirven la rueda del mouse y arrastrar)
  const botones = [
    ["+", () => zonaMapa.transition().call(zoom.scaleBy, 1.6)],
    ["−", () => zonaMapa.transition().call(zoom.scaleBy, 1 / 1.6)],
    ["⟲", () => zonaMapa.transition().call(zoom.transform, d3.zoomIdentity)]
  ];
  svg.append("g").selectAll("g").data(botones).join("g")
    .attr("transform", (d, i) => `translate(${mapaW - 30},${50 + i * 26})`)
    .style("cursor", "pointer")
    .on("click", (event, d) => d[1]())
    .call(g => g.append("rect").attr("width", 22).attr("height", 22)
                .attr("rx", 3).attr("fill", "white").attr("stroke", "#999"))
    .call(g => g.append("text").attr("x", 11).attr("y", 16)
                .attr("text-anchor", "middle").style("font-size", "14px").text(d => d[0]));

  // Leyenda de color VERTICAL (abajo a la izquierda, sobre el Pacífico)
  const legH = Math.min(140, mapaH * 0.3), legX = 10, legY = H - 15 - legH;
  // Recuadro blanco detrás de la leyenda para que se lea sobre el mapa con zoom
  svg.append("rect").attr("x", legX - 5).attr("y", legY - 25)
    .attr("width", 95).attr("height", legH + 35)
    .attr("rx", 4).attr("fill", "white").attr("stroke", "#bbb");
  const grad = DOM.uid("grad");
  svg.append("linearGradient").attr("id", grad.id)
    .attr("x1", 0).attr("y1", 1).attr("x2", 0).attr("y2", 0)  // degradado de abajo hacia arriba
    .selectAll("stop")
    .data(d3.ticks(0, 1, 10))
    .join("stop")
      .attr("offset", t => t)
      .attr("stop-color", t => color(color.domain()[0] + t * (color.domain()[1] - color.domain()[0])));
  svg.append("rect").attr("x", legX).attr("y", legY).attr("width", 10).attr("height", legH)
    .attr("fill", grad);
  svg.append("g").attr("transform", `translate(${legX + 10},0)`)
    .call(d3.axisRight(d3.scaleLinear().domain(color.domain()).range([legY + legH, legY]))
            .ticks(5).tickFormat(d => d + "%"))
    .call(g => g.select(".domain").remove());
  svg.append("text").attr("x", legX).attr("y", legY - 8).text("Cobertura total");

  // ===== 6. GRILLA 4x1 DE SERIES DE TIEMPO =====
  const [[, mapaY0], [, mapaY1]] = path.bounds(continente);
  const top = mapaY0, panelH = (mapaY1 - mapaY0) / niveles.length;
  const x = d3.scaleLinear().domain(d3.extent(munis, d => d.anio)).range([gx0, gx1]);

  // Leyenda de líneas
  const leyenda = [["Municipio seleccionado", ROJO], ["Departamento", AZUL], ["Otros municipios del depto.", "#aaa"]];
  svg.append("g").attr("transform", `translate(${gx0},18)`)
    .selectAll("g").data(leyenda).join("g")
      .attr("transform", (d, i) => `translate(${i * 175},0)`)
      .call(g => g.append("line").attr("x2", 18).attr("y1", -4).attr("y2", -4)
                  .attr("stroke", d => d[1]).attr("stroke-width", 3))
      .call(g => g.append("text").attr("x", 24).text(d => d[0]));

  const paneles = niveles.map((n, i) => {
    const y0 = top + i * panelH;
    const vMax = d3.quantile(munis.map(d => d[n.key]).filter(v => v != null), 0.995);
    const y = d3.scaleLinear().domain([0, Math.max(100, vMax)]).nice()
      .range([y0 + panelH - 22, y0 + 16]);
    const linea = d3.line()
      .defined(d => d[n.key] != null)
      .x(d => x(d.anio))
      .y(d => y(d[n.key]));
    const g = svg.append("g");

    // Ejes y título del panel
    g.append("g").attr("transform", `translate(0,${y.range()[0]})`)
      .call(d3.axisBottom(x).ticks(8).tickFormat(d3.format("d")));
    g.append("g").attr("transform", `translate(${gx0},0)`)
      .call(d3.axisLeft(y).ticks(4).tickFormat(d => d + "%"));
    g.append("text").attr("x", gx0 + 5).attr("y", y0 + 10)
      .style("font-weight", "bold").text(`Cobertura neta – ${n.titulo}`);
    const valores = g.append("text").attr("x", gx1).attr("y", y0 + 10)
      .attr("text-anchor", "end");

    // Línea de referencia 100%
    g.append("line").attr("x1", gx0).attr("x2", gx1)
      .attr("y1", y(100)).attr("y2", y(100))
      .attr("stroke", "#999").attr("stroke-dasharray", "3,3");

    // Recorte para que los outliers (>100%) no se salgan del panel
    const clip = DOM.uid("clip");
    g.append("clipPath").attr("id", clip.id)
      .append("rect").attr("x", gx0).attr("y", y.range()[1])
      .attr("width", gx1 - gx0).attr("height", y.range()[0] - y.range()[1]);
    const capa = g.append("g").attr("clip-path", clip);

    // Capa para las líneas grises (se llenan al seleccionar un municipio)
    const capaGris = capa.append("g");

    return {...n, linea, capa, capaGris, valores};
  });

  // ===== 7. Interacción: actualizar todo al hacer clic =====
  function actualizar(cod) {
    const serieM = seriesMuni.get(cod) ?? [];
    const cDpto = cod.slice(0, 2);
    const serieD = serieDepto(cDpto);
    const info = infoMuni.get(cod);
    const nombre = info ? info.entidad : featMuni.get(cod)?.properties.MPIO_CNMBR;
    const depto = info ? info.departamento : "";

    // Mapa
    // San Andrés (88) se resalta en el recuadro fijo; el resto en el mapa con zoom
    const [on, off] = cDpto === "88" ? [selIslas, selCont] : [selCont, selIslas];
    on.deptoSel.attr("d", featDepto.has(cDpto) ? pathMapa(featDepto.get(cDpto)) : null).raise();
    on.muniSel.attr("d", featMuni.has(cod) ? pathMapa(featMuni.get(cod)) : null).raise();
    off.deptoSel.attr("d", null);
    off.muniSel.attr("d", null);
    etiqueta.text(serieM.length ? `▶ ${nombre} (${depto})` : `▶ ${nombre}: sin datos`);

    // Los otros municipios del MISMO departamento (sin el seleccionado)
    const vecinos = Array.from(seriesMuni, ([c, s]) => ({cod: c, serie: s}))
      .filter(d => d.cod.slice(0, 2) === cDpto && d.cod !== cod);

    // Series: data binding con join() -> la línea entra, se actualiza o sale
    for (const p of paneles) {
      p.capaGris.selectAll("path")
        .data(vecinos, d => d.cod)          // la "key" identifica cada municipio
        .join(
          enter => enter.append("path")
            .attr("fill", "none").attr("stroke", GRIS)
            .attr("stroke-width", 1).attr("opacity", 0.8)
            .attr("d", d => p.linea(d.serie))
            .call(e => e.append("title").text(d => d.serie[0].entidad)),
          update => update,
          exit => exit.remove()
        );

      p.capa.selectAll("path.depto")
        .data([serieD])
        .join(enter => enter.append("path").attr("class", "depto")
                .attr("fill", "none").attr("stroke", AZUL).attr("stroke-width", 2))
        .transition().duration(500)
        .attr("d", p.linea);

      p.capa.selectAll("path.muni")
        .data([serieM])
        .join(enter => enter.append("path").attr("class", "muni")
                .attr("fill", "none").attr("stroke", ROJO).attr("stroke-width", 2.5))
        .raise()
        .transition().duration(500)
        .attr("d", p.linea);

      // Valores del último año en la esquina del panel
      const enUlt = s => s.find(d => d.anio === ultimoAnio)?.[p.key];
      const vm = enUlt(serieM), vd = enUlt(serieD);
      const f = v => (v == null ? "–" : v.toFixed(1) + "%");
      p.valores.text(`${ultimoAnio}: ${nombre} ${f(vm)} · ${depto || "Depto"} ${f(vd)}`);
    }
  }

  actualizar(inicial);
  return svg.node();
}
